# Camera
camera position 0 0.2 1.0
camera lookAt 0 0 -4
camera up 0 1 0
camera fov 60
camera near 0.01
camera far 1000
camera aperture 0.06
camera focusDistance 5.0

# Render
render width 320
render height 320
render background 0.02 0.02 0.03
render output output/dof.png
render dofSamples 12
render shadowSamples 2
render pointLightRadius 0.1

# Lights
light directional 0 -0.2 -1 1 1 1 0.3
light point 1.5 1.2 -1 1 1 1 15

# Materials
material nearRed 1 0.2 0.2 spec=0.4 shininess=32
material focusGreen 0.2 1 0.2 spec=0.4 shininess=32
material farBlue 0.2 0.3 1 spec=0.4 shininess=32

# Objects at different depths
sphere -1.2 0 -2.8 0.6 1 0.2 0.2 material=nearRed
sphere 0.0 0 -4.0 0.7 0.2 1 0.2 material=focusGreen
sphere 1.4 0 -6.2 0.9 0.2 0.3 1 material=farBlue
